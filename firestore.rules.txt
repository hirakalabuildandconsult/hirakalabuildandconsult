// Paste this into Firebase Console > Firestore Database > Rules, then click
// Publish. Replace you@example.com with the admin email you created in
// Authentication > Users (it must match adminEmail in js/firebase-config.js).
//
// What this does:
//  - Anyone can READ the content (so the public website can display it).
//  - Only someone signed in as your admin email can WRITE (add/edit/delete).

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null &&
             request.auth.token.email == "you@example.com";
    }

    match /projects/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /gallery/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /blogs/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /mcqQuestions/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /progressUpdates/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }
  }
}
